import { describe, it, expect, beforeAll, beforeEach, vi } from "vitest";
import { ParentDocument } from "@prisma-client";
import { MovementRepository } from "@app/common/repositories/movement.repository";
import { MovementService } from "@app/common/services/movement/movement.service";
import type { ClientProxy } from "@nestjs/microservices";
import { startLiveDb, type LiveDb } from "@test/postgres.container";
import { seedMovement, seedTradeItem, seedWarehouse, WS } from "@test/seed";

describe(`FIC-13173 — [Lista movimenti] Viene mostrato solo il numero documento e non la tipologia`, () => {
  let db: LiveDb;
  let service: MovementService;

  beforeAll(async () => {
    db = await startLiveDb();
  });

  beforeEach(async () => {
    await db.restore();
    service = new MovementService(
      { emit: vi.fn() } as unknown as ClientProxy,
      new MovementRepository(db.prisma()),
      { get: vi.fn(), set: vi.fn(), clear: vi.fn() } as never,
      { createFromEventDTO: vi.fn() } as never,
      { get: vi.fn() } as never,
      { get: vi.fn() } as never,
      { add: vi.fn() } as never,
    );
  });

  // Documento filter (service → real repo): typology travels with the number
  it("getAllDocuments returns { document, parentDocument }[] with total via real repo", async () => {
    const prisma = db.prisma();
    await seedWarehouse(prisma);
    await seedTradeItem(prisma);
    await seedMovement(prisma, {
      eventId: "evt-inv",
      document: "78",
      parentDocument: ParentDocument.purchaseInvoice,
    });
    await seedMovement(prisma, {
      eventId: "evt-ord",
      document: "98",
      parentDocument: ParentDocument.purchaseOrder,
    });
    await seedMovement(prisma, {
      eventId: "evt-adj",
      document: "ADJ-9",
      parentDocument: ParentDocument.inventoryStockAdjustmentNote,
    });

    const res = await service.getAllDocuments(WS);
    type DocRow = { document: string; parentDocument: string | null };
    const rows = res.data as unknown as DocRow[];

    expect(res.total).toBe(2);
    expect(rows).toEqual(
      expect.arrayContaining([
        { document: "78", parentDocument: ParentDocument.purchaseInvoice },
        { document: "98", parentDocument: ParentDocument.purchaseOrder },
      ]),
    );
    expect(rows).toHaveLength(2);
    expect(
      rows.some(
        (d) => d.parentDocument === ParentDocument.inventoryStockAdjustmentNote,
      ),
    ).toBe(false);
  });
});

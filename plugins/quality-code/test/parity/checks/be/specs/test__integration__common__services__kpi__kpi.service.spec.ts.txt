import { describe, it, expect, beforeAll, beforeEach, vi } from "vitest";
import {
  MovementType,
  MovementReason,
  Operation,
  EventSource,
} from "@prisma-client";
import { KpiService } from "@app/common/services/kpi/kpi.service";
import { KpiRepository } from "@app/common/repositories/kpi.repository";
import type { EventDTO } from "@app/common/dto/event/event.dto";
import { EventType } from "@app/common/enums/event/type.enum";
import { startLiveDb, type LiveDb } from "@test/postgres.container";
import { seedTradeItem, seedKpi, seedMovement, WS, TI, WH } from "@test/seed";
import { allocationService, cacheDouble } from "@test/fakes";

// FIC-13436 — service-level view of the stock-reconstruction bug: an OUTGOING movement must
// DECREASE stockOnHand. KpiService.processKpiUpdate → handleMovement → calculateStockDelta signs
// the quantity from the stock config's stockOnHandOperation (decrement for outgoing), then the real
// KpiRepository.upsertKpiStock accumulates it. If the outgoing sign were dropped/inverted (the
// 116.75 defect), stock would grow. Two real collaborators (KpiService + real KpiRepository + Prisma).
describe(`FIC-13436 — [<secret:5>] Incorrect warehouse stock`, () => {
  let db: LiveDb;
  let service: KpiService;

  beforeAll(async () => {
    db = await startLiveDb();
  });

  beforeEach(async () => {
    await db.restore();
    service = new KpiService(
      new KpiRepository(db.prisma()),
      // eventsService: store returns a truthy stored event so processing continues
      {
        createFromEventDTO: vi.fn().mockResolvedValue({ eventId: "evt-kpi-13436" }),
      } as never,
      // stockCalculationConfigurationsService: outgoing → decrement stock on hand
      {
        get: vi.fn().mockResolvedValue({
          stockOnHandOperation: Operation.decrement,
          stockAvailabilityOperation: Operation.decrement,
          stockValueOperation: Operation.decrement,
          orderedStockOperation: null,
          reservedStockOperation: null,
        }),
      } as never,
      allocationService(db.prisma(), // stockCalculationConfigurationsService: outgoing → decrement stock on hand
      {
        get: vi.fn().mockResolvedValue({
          stockOnHandOperation: Operation.decrement,
          stockAvailabilityOperation: Operation.decrement,
          stockValueOperation: Operation.decrement,
          orderedStockOperation: null,
          reservedStockOperation: null,
        }),
      } as never),
      cacheDouble() as never,
    );
  });

  it("an outgoing movement subtracts the uncorrected 116.75 down to 58.86 (never grows)", async () => {
    const prisma = db.prisma();
    await seedTradeItem(prisma);
    await seedKpi(prisma, { stockOnHand: "116.75", date: new Date("2024-01-01") });

    // The allocation service persists a snapshot onto the movement row, so the row the payload
    // names has to exist. Seed it first and reuse its real id.
    const row = await seedMovement(prisma, {
      eventId: "evt-kpi-13436",
      type: MovementType.outgoing,
      reason: MovementReason.sale,
      quantity: "57.89",
      unitPrice: "10",
      totalAmount: "578.9",
      date: new Date("2024-01-02"),
    });

    const movement = {
      id: row.id,
      workspaceId: WS,
      tradeItemId: TI,
      warehouseId: WH,
      type: MovementType.outgoing,
      reason: MovementReason.sale,
      quantity: "57.89",
      unitPrice: "10",
      date: "2024-01-02",
    };
    const event = {
      eventId: "evt-kpi-13436",
      eventType: EventType.MOVEMENT_GENERATION,
      source: EventSource.manual,
      workspaceId: WS,
      payload: JSON.stringify(movement),
    } as unknown as EventDTO;

    await service.processKpiUpdate(event);

    const kpi = await prisma.client.kpi.findFirst({
      where: {
        workspaceId: WS,
        tradeItemId: TI,
        warehouseId: WH,
        date: new Date("2024-01-02"),
      },
    });

    expect(kpi).not.toBeNull();
    expect(kpi!.stockOnHand.toNumber()).toBeCloseTo(58.86, 2);
    expect(kpi!.stockOnHand.toNumber()).not.toBeCloseTo(116.75, 2);
  });
});

// FIC-13986 — [U######] Incorrect stock between report and product master data.
// The customer's stock report (report di giacenza) showed 3 fewer units than the product master
// data (anagrafica) for two wines: "Carnasciale 2020" should read 914, "Ottantadue 2022" should
// read 77. Both numbers come off the same KPI snapshot table but through two paths:
//   master data → the latest KPI row for the item  (TradeItem.get: kpi orderBy date desc, take 1)
//   report      → KpiService.getStockDataForDateWithFallback: latest KPI ≤ selectedDate, per warehouse
// This anchor drives the write path (processKpiUpdate → calculateStockDelta → upsertKpiStock) with
// real movements, then asserts the two read paths AGREE and equal the true net total. If the sign
// logic or the snapshot read drifts, the report would fall below the master data again.
describe(`FIC-13986 — [U######] Incorrect stock between report and product master data`, () => {
  let db: LiveDb;
  let service: KpiService;

  const CARNASCIALE = "ti-carnasciale-2020";
  const OTTANTADUE = "ti-ottantadue-2022";
  // getStockDataForDateWithFallback filters `trade_items.created_at <= selectedDate`, and seeded
  // rows are created "now". Keep the KPI date at now and ask for the report a week out, so the
  // as-of window always covers the seeded rows no matter when the suite runs.
  const MOV_DATE = new Date();
  const AS_OF = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);

  beforeAll(async () => {
    db = await startLiveDb();
  });

  beforeEach(async () => {
    await db.restore();
    service = new KpiService(
      new KpiRepository(db.prisma()),
      {
        createFromEventDTO: vi.fn().mockResolvedValue({ eventId: "evt-13986" }),
      } as never,
      // incoming increases stock on hand, outgoing decreases it (the real config's direction)
      {
        get: vi.fn().mockImplementation((m: { type: MovementType }) => {
          const op =
            m.type === MovementType.incoming
              ? Operation.increment
              : Operation.decrement;
          return Promise.resolve({
            stockOnHandOperation: op,
            stockAvailabilityOperation: op,
            stockValueOperation: op,
            orderedStockOperation: null,
            reservedStockOperation: null,
          });
        }),
      } as never,
      allocationService(db.prisma(), {
        get: vi.fn().mockImplementation((m: { type: MovementType }) => {
          const op =
            m.type === MovementType.incoming
              ? Operation.increment
              : Operation.decrement;
          return Promise.resolve({
            stockOnHandOperation: op,
            stockAvailabilityOperation: op,
            stockValueOperation: op,
            orderedStockOperation: null,
            reservedStockOperation: null,
          });
        }),
      }),
      cacheDouble() as never,
    );
  });

  async function applyMovement(
    tradeItemId: string,
    type: MovementType,
    quantity: string,
    seq: number,
  ): Promise<void> {
    // The allocation service writes a snapshot onto the movement row, so seed the row the
    // payload names and reuse its real id.
    const row = await seedMovement(db.prisma(), {
      eventId: `evt-13986-${tradeItemId}-${seq}`,
      tradeItemId,
      type,
      reason:
        type === MovementType.incoming
          ? MovementReason.positiveAdjustment
          : MovementReason.sale,
      quantity,
      unitPrice: "10",
      date: MOV_DATE,
    });

    const movement = {
      id: row.id,
      workspaceId: WS,
      tradeItemId,
      warehouseId: WH,
      type,
      reason:
        type === MovementType.incoming
          ? MovementReason.positiveAdjustment
          : MovementReason.sale,
      quantity,
      unitPrice: "10",
      date: MOV_DATE.toISOString(),
    };
    await service.processKpiUpdate({
      eventId: `evt-13986-${tradeItemId}-${seq}`,
      eventType: EventType.MOVEMENT_GENERATION,
      source: EventSource.manual,
      workspaceId: WS,
      payload: JSON.stringify(movement),
    } as unknown as EventDTO);
  }

  /** The product-master-data value: the latest KPI row for the item (TradeItem.get, take 1). */
  async function masterStock(tradeItemId: string): Promise<number> {
    const kpi = await db.prisma().client.kpi.findFirst({
      where: { workspaceId: WS, tradeItemId, warehouseId: WH },
      orderBy: { date: "desc" },
    });
    return kpi!.stockOnHand.toNumber();
  }

  /** The report value: KpiService.getStockDataForDateWithFallback (latest KPI ≤ selectedDate). */
  async function reportStock(tradeItemId: string): Promise<number> {
    const rows = await service.getStockDataForDateWithFallback(WS, WH, AS_OF);
    return rows.find((r) => r.tradeItemId === tradeItemId)!.stockOnHand.toNumber();
  }

  it("report giacenza equals the master-data stock: 914 for Carnasciale 2020, 77 for Ottantadue 2022", async () => {
    const prisma = db.prisma();
    await seedTradeItem(prisma, {
      id: CARNASCIALE,
      code: "Carnasciale 2020",
      name: "Carnasciale 2020",
    });
    await seedTradeItem(prisma, {
      id: OTTANTADUE,
      code: "Ottantadue 2022",
      name: "Ottantadue 2022",
    });

    // 917 in, 3 out → 914 ;  80 in, 3 out → 77  (the ticket's -3 discrepancy shape)
    await applyMovement(CARNASCIALE, MovementType.incoming, "917", 1);
    await applyMovement(CARNASCIALE, MovementType.outgoing, "3", 2);
    await applyMovement(OTTANTADUE, MovementType.incoming, "80", 1);
    await applyMovement(OTTANTADUE, MovementType.outgoing, "3", 2);

    const carnascialeReport = await reportStock(CARNASCIALE);
    const ottantadueReport = await reportStock(OTTANTADUE);

    // the ticket's expected values — the report must show the correct master-data stock
    expect(carnascialeReport).toBe(914);
    expect(ottantadueReport).toBe(77);
    // and it must equal the product master data (anagrafica), not fall 3 short of it
    expect(carnascialeReport).toBe(await masterStock(CARNASCIALE));
    expect(ottantadueReport).toBe(await masterStock(OTTANTADUE));
  });

  it("every item in the report matches its master-data stock (no report-vs-anagrafica drift)", async () => {
    const prisma = db.prisma();
    await seedTradeItem(prisma, {
      id: CARNASCIALE,
      code: "Carnasciale 2020",
      name: "Carnasciale 2020",
    });
    await seedTradeItem(prisma, {
      id: OTTANTADUE,
      code: "Ottantadue 2022",
      name: "Ottantadue 2022",
    });

    await applyMovement(CARNASCIALE, MovementType.incoming, "917", 1);
    await applyMovement(CARNASCIALE, MovementType.outgoing, "3", 2);
    await applyMovement(OTTANTADUE, MovementType.incoming, "80", 1);
    await applyMovement(OTTANTADUE, MovementType.outgoing, "3", 2);

    const rows = await service.getStockDataForDateWithFallback(WS, WH, AS_OF);
    for (const row of rows) {
      expect(row.stockOnHand.toNumber()).toBe(await masterStock(row.tradeItemId));
    }
    // the two pinned items resolve to the ticket's numbers
    expect(
      rows.find((r) => r.tradeItemId === CARNASCIALE)!.stockOnHand.toNumber(),
    ).toBe(914);
    expect(
      rows.find((r) => r.tradeItemId === OTTANTADUE)!.stockOnHand.toNumber(),
    ).toBe(77);
  });
});

const PRIMARY = wh('wh-primary', 'Magazzino Centrale', 'MC', {
  isPrimary: true,

// Tag for tests that wait on the multi-warehouse feature (see the header).
const ANCHOR = { tag: ['@anchor', '@FIC-13374'] }

// A warehouse card, found by the warehouse name (unique per card).
const card = (page: Page, name: string) =>
  page.locator('.MuiCard-root').filter({ hasText: name })
// The delete confirmation window.
const dialog = (page: Page) => page.getByRole('dialog')
// The delete icon is the middle control of a card's action row (Edit, Delete, "…").
const deleteIcon = (page: Page, name: string) =>
  card(page, name).getByRole('button').nth(1)

interface StubOptions {
  active: TestWarehouse[]
  archived?: TestWarehouse[]
  deleteStatus?: number
  bucket?: unknown
}

async function stubWarehouses(page: Page, opts: StubOptions) {
  const { active, archived = [], deleteStatus = 204, bucket = IT_BUCKET } = opts
  let activeList = [...active]
  let archivedList = [...archived]

  await page.route('**/i18n/**', route =>
    route.fulfill({ json: { item: bucket } })
  )
  await page.route('**/warehouse/workspace/*/active', route =>
    route.fulfill({ json: activeList })
  )
  await page.route('**/warehouse/workspace/*/archived', route =>
    route.fulfill({ json: archivedList })
  )
  // DELETE warehouse/{id}/workspace/{ws}. On success the deleted warehouse really leaves the
  // list, so the refetch the page runs afterwards removes its card for real.
  await page.route('**/warehouse/*/workspace/**', async route => {
    if (route.request().method() !== 'DELETE') return route.fallback()
    const id = decodeURIComponent(route.request().url()).match(
      /warehouse\/([^/]+)\/workspace/
    )?.[1]
    if (deleteStatus >= 200 && deleteStatus < 300) {
      activeList = activeList.filter(w => w.id !== id)
      archivedList = archivedList.filter(w => w.id !== id)
    }
    await route.fulfill({
      status: deleteStatus,
      contentType: 'application/json',
      body: '{}',
    })
  })
}

// --- Acceptance criteria (from the ticket) ---

// Scenario: delete icon is available for an eligible warehouse
//   Given a warehouse has no movements, is not the primary warehouse, and at least 2 exist
//   When I open the warehouse management grid
//   Then that warehouse's card shows a delete icon
test(
  'FIC-13374-1-eligible-warehouse-card-shows-a-delete-icon',
  ANCHOR,
  async ({ page, warehouse }) => {
    await stubWarehouses(page, { active: [PRIMARY, NORTH] })
    await warehouse.goto('/warehouseManagement')

    const eligible = card(page, NORTH.name)
    await expect(eligible).toBeVisible({ timeout: 15_000 })
    // Edit, Delete and the "…" menu — three controls; the delete icon only appears when eligible.
    await expect(eligible.getByRole('button')).toHaveCount(3)
    // Prove the middle control really is delete: it opens the deletion window.
    await deleteIcon(page, NORTH.name).click()
    await expect(dialog(page).getByText(IT.modalText)).toBeVisible()
  }
)

// Scenario: delete icon is hidden for an ineligible warehouse — the primary one
//   Given a warehouse is the primary warehouse
//   When I open the warehouse management grid
//   Then that warehouse's card does not show a delete icon
test(
  'FIC-13374-2-primary-warehouse-card-has-no-delete-icon',
  ANCHOR,
  async ({ page, warehouse }) => {
    await stubWarehouses(page, { active: [PRIMARY, NORTH] })
    await warehouse.goto('/warehouseManagement')

    const primary = card(page, PRIMARY.name)
    await expect(primary).toBeVisible({ timeout: 15_000 })
    // The primary warehouse offers only Edit: no delete, and no "…" menu (it cannot be archived
    // or set primary again).
    await expect(primary.getByRole('button')).toHaveCount(1)
  }
)

// Scenario: delete icon is hidden for an ineligible warehouse — the only one left
//   Given only 1 warehouse exists
//   When I open the warehouse management grid
//   Then that warehouse's card does not show a delete icon
test(
  'FIC-13374-3-the-only-warehouse-has-no-delete-icon',
  ANCHOR,
  async ({ page, warehouse }) => {
    const only = wh('wh-only', 'Magazzino Unico', 'MU')
    await stubWarehouses(page, { active: [only] })
    await warehouse.goto('/warehouseManagement')

    const onlyCard = card(page, only.name)
    await expect(onlyCard).toBeVisible({ timeout: 15_000 })
    // Edit and the "…" menu, but no delete: a warehouse cannot be deleted when it is the last one.
    await expect(onlyCard.getByRole('button')).toHaveCount(2)
  }
)

// Scenario: delete icon is hidden for a warehouse that has movements
//   The ticket lists "has at least one movement" as a reason to hide the delete icon. The shipped
//   code decides eligibility from `canDelete = !isPrimary && totalWarehouses >= 2` only — it never
//   looks at movements, and the warehouse list has no movements / "deletable" flag to drive this
//   from. So this rule cannot be exercised from the screen yet. The acceptance criteria and the
//   implementation disagree; this is a finding for the team, not something to assert on today.
test.fixme('FIC-13374-4-warehouse-with-movements-has-no-delete-icon', async () => {})

// Scenario: confirming deletion removes the warehouse
//   Given I click the delete icon on an eligible warehouse's card
//   When the "Eliminazione magazzino" modal opens and I click "Elimina magazzino"
//   Then the card is removed, a success toast "Magazzino eliminato" shows, and I stay on the tab
test(
  'FIC-13374-5-confirming-removes-the-warehouse-and-shows-a-success-message',
  ANCHOR,
  async ({ page, warehouse }) => {
    // Three active warehouses, so one deletion still leaves two and the tab stays visible.
    await stubWarehouses(page, { active: [PRIMARY, NORTH, SOUTH] })
    await warehouse.goto('/warehouseManagement')

    await expect(card(page, NORTH.name)).toBeVisible({ timeout: 15_000 })
    await deleteIcon(page, NORTH.name).click()
    await expect(dialog(page).getByText(IT.modalText)).toBeVisible()
    await dialog(page).getByRole('button', { name: IT.confirm }).click()

    // Success message is shown.
    await expect(page.getByText(IT.success)).toBeVisible({ timeout: 15_000 })
    // The card is gone from the grid.
    await expect(page.getByText(NORTH.name)).toHaveCount(0)
    // We are still on the first (active) tab we started from.
    await expect(page.getByRole('tab').first()).toHaveAttribute(
      'aria-selected',
      'true'
    )
  }
)

// Scenario: cancelling the modal keeps the warehouse
//   Given the "Eliminazione magazzino" modal is open
//   When I click "Annulla"
//   Then the modal closes and the warehouse card is still in the grid
test(
  'FIC-13374-6-cancelling-the-window-keeps-the-warehouse',
  ANCHOR,
  async ({ page, warehouse }) => {
    await stubWarehouses(page, { active: [PRIMARY, NORTH] })
    await warehouse.goto('/warehouseManagement')

    await expect(card(page, NORTH.name)).toBeVisible({ timeout: 15_000 })
    await deleteIcon(page, NORTH.name).click()
    // The window explains the deletion is permanent.
    await expect(dialog(page).getByText(IT.modalText)).toBeVisible()
    await dialog(page).getByRole('button', { name: IT.cancel }).click()

    await expect(dialog(page)).toHaveCount(0)
    await expect(card(page, NORTH.name)).toBeVisible()
  }
)

// Scenario: a failed deletion shows an error and keeps the warehouse
//   Given I click "Elimina magazzino" and the delete request fails
//   When the response returns
//   Then an error toast "Errore nell'eliminazione del magazzino, riprova" shows and the card stays
test(
  'FIC-13374-7-a-failed-deletion-shows-an-error-and-keeps-the-warehouse',
  ANCHOR,
  async ({ page, warehouse }) => {
    await stubWarehouses(page, { active: [PRIMARY, NORTH], deleteStatus: 500 })

    await expect(card(page, NORTH.name)).toBeVisible({ timeout: 15_000 })
    await deleteIcon(page, NORTH.name).click()
    await dialog(page).getByRole('button', { name: IT.confirm }).click()

    await expect(page.getByText(IT.error)).toBeVisible({ timeout: 15_000 })
    await expect(card(page, NORTH.name)).toBeVisible()
  }
)

// E5 — large list: a tenant with many warehouses must still render the grid without hanging.
test(
  'FIC-13374-edge-5-a-long-list-of-warehouses-still-loads',
  ANCHOR,
  async ({ page, warehouse }) => {
    const many = [
      PRIMARY,
      ...Array.from({ length: 119 }, (_, i) =>
        wh(`wh-${i}`, `Magazzino ${String(i).padStart(3, '0')}`, `W${i}`)
      ),
    ]
    await stubWarehouses(page, { active: many })
    await warehouse.goto('/warehouseManagement')

    // The grid renders all 120 cards (119 warehouses + the primary), no hang or blank screen.
    await expect(page.locator('.MuiCard-root')).toHaveCount(120, {
      timeout: 15_000,
    })
  }
)
